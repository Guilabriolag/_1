// ── STORAGE ──
function loadFromStorage() {
  const saved = localStorage.getItem('labsystem_data');
  if (saved) {
    const d = JSON.parse(saved);
    storeData = d.store || {};
    products = d.products || getSampleProducts();
    keys = d.keys || [];
    orders = d.orders || getSampleOrders();
    cupons = d.cupons || [];
  } else {
    products = getSampleProducts();
    orders = getSampleOrders();
  }
  if (storeData.nome) { 
    document.getElementById('c-nome').value = storeData.nome;
    document.getElementById('drawer-store-name').textContent = storeData.nome;
  }
  if (storeData.slogan) document.getElementById('c-slogan').value = storeData.slogan;
  if (storeData.whatsapp) document.getElementById('cfg-whatsapp').value = storeData.whatsapp;
}

function saveToStorage() {
  localStorage.setItem('labsystem_data', JSON.stringify({ store: storeData, products, keys, orders, cupons }));
}

}

function renderDashboard() {
  const total = orders.reduce((s, o) => s + (o.total || 0), 0);
  document.getElementById('stat-vendas').textContent = 'R$ ' + total.toFixed(2).replace('.', ',');
  document.getElementById('stat-pedidos').textContent = orders.length;
  const baixo = products.filter(p => p.estoque <= p.alerta).length;
  document.getElementById('stat-estoque').textContent = baixo;
  renderRecentOrdersMobile();
}

function renderMiniChart() {
  const dias = ['S','T','Q','Q','S','S','D'];
  const vals = dias.map((_, i) => Math.floor(Math.random() * 70) + 20);
  const max = Math.max(...vals);
  const chart = document.getElementById('mini-chart');
  const labels = document.getElementById('mini-labels');
  chart.innerHTML = vals.map((v, i) => 
    `<div class="mini-bar ${i === 6 ? 'today' : ''}" style="height:${(v/max)*100}%" title="${dias[i]}"></div>`
  ).join('');
  labels.innerHTML = dias.map(d => `<div style="flex:1;text-align:center;font-size:9px;color:var(--text3);font-family:var(--font-mono);">${d}</div>`).join('');
}

function renderTopProdsMobile() {
  const el = document.getElementById('top-prods-mobile');
  const sorted = [...products].sort((a, b) => (b.vendas||0) - (a.vendas||0)).slice(0, 4);
  const max = sorted[0]?.vendas || 1;
  el.innerHTML = `<div style="padding:0 16px 8px;">` + sorted.map(p => `
    <div style="margin-bottom:14px;">
      <div style="display:flex;justify-content:space-between;font-size:13px;margin-bottom:5px;">
        <span style="color:var(--text);font-weight:500;">${p.img} ${p.nome}</span>
        <span style="font-family:var(--font-mono);color:var(--accent3);font-size:12px;">${p.vendas||0}x</span>
      </div>
      <div class="progress"><div class="progress-fill g" style="width:${((p.vendas||0)/max)*100}%"></div></div>
    </div>`).join('') + `</div>`;
}

function gerarQR() {
  const url = document.getElementById('qr-url').value.trim();
  const mode = document.getElementById('qr-mode').value;
  const qty = parseInt(document.getElementById('qr-qty').value) || 1;
  const color = document.getElementById('qr-color').value;
  const bg = document.getElementById('qr-bg').value;

  if (!url && mode === 'manual') { notify('Preencha a URL!', 'error'); return; }

  let finalUrl = url;
  if (mode === 'numerado') finalUrl = url + (url.endsWith('=') ? '' : '?src=') + String(Math.floor(Math.random()*99)+1).padStart(2,'0');
  else if (mode === 'letras') finalUrl = url + (url.endsWith('=') ? '' : '?src=') + String.fromCharCode(65 + Math.floor(Math.random()*26));
  else if (mode === 'aleatorio') {
    const r = Math.random().toString(36).substring(2,8).toUpperCase();
    finalUrl = url + (url.endsWith('=') ? '' : '?src=') + r;
  }

  // Generate QR using Google Charts API (no lib needed)
  const wrap = document.getElementById('qr-canvas-wrap');
  const encoded = encodeURIComponent(finalUrl);
  const qrUrl = `https://api.qrserver.com/v1/create-qr-code/?size=180x180&data=${encoded}&color=${color.replace('#','')}&bgcolor=${bg.replace('#','')}`;
  wrap.innerHTML = `<img src="${qrUrl}" width="150" height="150" style="border-radius:6px;display:block;">`;

  document.getElementById('qr-url-display').textContent = finalUrl;
  document.getElementById('qr-preview-card').style.display = 'block';

  // HMAC signature
  const hmacKey = document.getElementById('qr-hmac-key').value;
  if (hmacKey) {
    gerarHMAC(finalUrl, hmacKey);
  }

  notify(`QR Code gerado! (${qrCurrentType.toUpperCase()})`, 'success');
}

async function gerarHMAC(data, key) {
  try {
    const enc = new TextEncoder();
    const cryptoKey = await crypto.subtle.importKey('raw', enc.encode(key), {name:'HMAC',hash:'SHA-256'}, false, ['sign']);
    const sig = await crypto.subtle.sign('HMAC', cryptoKey, enc.encode(data));
    const hex = Array.from(new Uint8Array(sig)).map(b => b.toString(16).padStart(2,'0')).join('');
    document.getElementById('qr-sig-value').textContent = hex;
    document.getElementById('qr-signature-result').style.display = 'block';
  } catch(e) { console.warn('HMAC:', e); }
}

async function hashString(str) {
  const buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(str));
  return Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2,'0')).join('');
}

function fakebtcTx() {
  return Array.from({length:64}, ()=>'0123456789abcdef'[Math.floor(Math.random()*16)]).join('');
}

async function registrarLabChain() {
  const tipo = document.getElementById('chain-doc-type').value;
  const partes = document.getElementById('chain-partes').value.trim();
  const cidade = document.getElementById('chain-cidade').value;
  const conteudo = document.getElementById('chain-content').value.trim();

  if (!partes || !conteudo) { notify('Preencha todos os campos!', 'error'); return; }

  const ts = new Date().toISOString();
  const docStr = `${tipo}|${partes}|${cidade}|${conteudo}|${ts}`;
  const docHash = await hashString(docStr);
  const btcTx = fakebtcTx();

  // Cert URL
  const certUrl = `https://labriolag.shop/chain/verify?h=${docHash.substring(0,16)}&t=${Date.now()}`;

  // Display
  document.getElementById('chain-doc-hash').textContent = docHash;
  document.getElementById('chain-timestamp').textContent = new Date().toLocaleString('pt-BR') + ' · Hash ID: ' + docHash.substring(0,8).toUpperCase();
  document.getElementById('chain-btc-tx').textContent = btcTx + ' (1 Sat → LabSocial Pirapora)';

  // QR do certificado
  const wrap = document.getElementById('chain-qr-wrap');
  wrap.innerHTML = `<img src="https://api.qrserver.com/v1/create-qr-code/?size=120x120&data=${encodeURIComponent(certUrl)}&color=f5a623&bgcolor=ffffff" width="100" height="100" style="border-radius:4px;">`;

  document.getElementById('chain-result-card').style.display = 'block';

  // Save to history
  const history = JSON.parse(localStorage.getItem('labchain_history') || '[]');
  const entry = { tipo, partes, cidade, hash: docHash, tx: btcTx, ts };
  history.unshift(entry);
  localStorage.setItem('labchain_history', JSON.stringify(history.slice(0,20)));

  renderChainHistory();
  updateChainStats();
  notify('Documento registrado na LabChain! 🔐', 'success');
}
