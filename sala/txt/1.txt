  castelo_branco: { code: "SP-280", color: 0xe6c34f, nodes: ["itapevi", "barueri", "alphaville", "aracariguama", "sao_roque"] },

// Centralizando em Barueri para manter números menores no 3D
const centerLat = GEO_NODES.barueri.lat;
const centerLon = GEO_NODES.barueri.lon;

function lonToMeters(lon) {
    return (lon - centerLon) * 111320 * Math.cos(centerLat * Math.PI / 180);
}
function latToMeters(lat) {
    return (lat - centerLat) * 111320;
}

// Converter nós geográficos para nós 3D (X, Z) - Three.js usa Y para cima
const NODES_3D = {};
Object.keys(GEO_NODES).forEach(key => {
    NODES_3D[key] = new THREE.Vector3(
        lonToMeters(GEO_NODES[key].lon),
        0, // Altura base
        -latToMeters(GEO_NODES[key].lat) // Z invertido no Three.js em relação ao norte geográfico
    );

const northM = latToMeters(Math.max(...lats)) + PAD_KM * 1000;

const eastM  = lonToMeters(Math.max(...lons)) + PAD_KM * 1000;

const mapHeightM = northM - southM;

function init3D() {
    // 1. Cena
    scene = new THREE.Scene();
    scene.background = new THREE.Color(0x050706);
    // Neblina para dar sensação de profundidade
    scene.fog = new THREE.FogExp2(0x050706, 0.00002);

    // 2. Câmera Isométrica (Orthographic)
    const aspect = window.innerWidth / window.innerHeight;
    const d = 15000; // Tamanho do frustum (área visível) em metros

    // Posição clássica isométrica: 45 graus em y, ~35 graus em x

    controls.enableZoom = true;
    controls.enablePan = true;
    controls.target.set(mapCenterX, 0, mapCenterZ); // Focar no centro do mapa
    controls.update();

    // 5. Iluminação

// ---------- DESENHAR ESTRADAS 3D ----------
function build3DRoads() {
    roadsGroup.clear();
    
    Object.keys(ROAD_NETWORK).forEach(key => {
        const road = ROAD_NETWORK[key];
        const points = road.nodes.map(nodeKey => NODES_3D[nodeKey]);
        
        // Criar tubo ou linha grossa para a estrada
        const curve = new THREE.CatmullRomCurve3(points);
        const geometry = new THREE.TubeGeometry(curve, 64, 40, 8, false); // Tubo de 40m de raio
        const material = new THREE.MeshBasicMaterial({ color: road.color, opacity: 0.8, transparent: true });
        const tube = new THREE.Mesh(geometry, material);
        tube.position.y = 50; // Levemente acima do chão
        roadsGroup.add(tube);
    });
}

// ---------- DESENHAR GRID DE CHUNKS 3D (A "OUTRA CAMADA HORIZONTAL/VERTICAL") ----------

// Utilitário matemática para distância ponto-segmento plano (XZ)
function distPointToSegmentSq(p, v, w) {
  const l2 = v.distanceToSquared(w);
  if (l2 === 0) return p.distanceToSquared(v);
  let t = ((p.x - v.x) * (w.x - v.x) + (p.z - v.z) * (w.z - v.z)) / l2;
  t = Math.max(0, Math.min(1, t));
  return p.distanceToSquared(new THREE.Vector3(v.x + t * (w.x - v.x), 0, v.z + t * (w.z - v.z)));
}

function build3DGrid(sizeKM) {
    gridGroup.clear();
    interactiveObjects = []; // Limpar para novo cálculo
    
    const sizeM = sizeKM * 1000;
    const cols = Math.ceil(mapWidthM / sizeM);
    const rows = Math.ceil(mapHeightM / sizeM);

    document.getElementById('chunkCount').textContent = (cols * rows).toLocaleString('pt-BR');
    document.getElementById('chunkDims').textContent = `${cols} × ${rows}`;

    // Geometrias reutilizáveis (Cubos/Box)
    const geomVazio = new THREE.BoxGeometry(sizeM * 0.95, 200, sizeM * 0.95); // Mais baixo
    const geomVia = new THREE.BoxGeometry(sizeM * 0.98, 800, sizeM * 0.98); // Mais alto

    // Materiais
    const matVazio = new THREE.MeshPhongMaterial({ 
        color: 0x4fd1a5, 
        transparent: true, 
        opacity: 0.1, 
        wireframe: false,
        emissive: 0x4fd1a5,
        emissiveIntensity: 0.1
    });
    const matVazioWire = new THREE.LineBasicMaterial({ color: 0x2f7a5c, opacity: 0.3, transparent: true });

    const matVia = new THREE.MeshPhongMaterial({ 
        color: 0xffb347, 
        transparent: true, 
        opacity: 0.6,
        emissive: 0xffb347,
        emissiveIntensity: 0.2
    });

    // Threshold de detecção de via (metade da diagonal do chunk)
    const distThresholdSq = Math.pow(sizeM * 0.7, 2);

    // Iterar para criar blocos
    for (let ix = 0; ix < cols; ix++) {
        for (let iz = 0; iz < rows; iz++) {
            
            // Posição central do chunk no mundo 3D
            const centerX = westM + (ix * sizeM) + sizeM / 2;
            const centerZ = -(southM + (iz * sizeM) + sizeM / 2); // Z Three.js
            const centerPos = new THREE.Vector3(centerX, 0, centerZ);

            // Verificação de proximidade com a rede viária
            let onRoute = false;
            for (const roadKey in ROAD_NETWORK) {
                const road = ROAD_NETWORK[roadKey];
                for (let i = 0; i < road.nodes.length - 1; i++) {
                    const v = NODES_3D[road.nodes[i]];
                    const w = NODES_3D[road.nodes[i+1]];
                    if (distPointToSegmentSq(centerPos, v, w) < distThresholdSq) {
                        onRoute = true;
                        break;
                    }
                }
                if (onRoute) break;
            }

            // Criar a malha (Mesh) baseada no status
            let block;
            if (onRoute) {
                block = new THREE.Mesh(geomVia, matVia);
                block.position.set(centerX, 400, centerZ); // Centralizado na altura (800/2)
            } else {
                block = new THREE.Mesh(geomVazio, matVazio);
                block.position.set(centerX, 100, centerZ); // Centralizado na altura (200/2)
                
                // Adicionar wireframe sutil para blocos vazios
                const wireframe = new THREE.LineSegments(new THREE.EdgesGeometry(geomVazio), matVazioWire);
                wireframe.position.copy(block.position);
                gridGroup.add(wireframe);
            }

            // Dados customizados para interação
            block.userData = {
                id: `[${ix}, ${iz}]`,
                posX: centerX.toFixed(0),
                posZ: centerZ.toFixed(0),
                status: onRoute ? 'Atravessado por via' : 'Vazio'
            };

            gridGroup.add(block);
            interactiveObjects.push(block); // Adicionar para detecção de clique
        }
    }
}

Agora sim. Olhando os dois códigos juntos, dá para ver exatamente onde você chegou.

Você tem, na verdade, duas camadas diferentes do projeto:

1. O primeiro código é o CHUNK

Ele pega o espaço real e transforma em uma malha espacial:

mundo ↓ região ↓ grid ↓ chunks 

E o Three.js já está usando esses chunks como volumes 3D.

Só que ainda existe uma lógica de “chunk de mapa”: ele está pensando principalmente em área, estrada e posição geográfica.

2. O segundo código é o começo do INDEX SPACE

Esse é mais importante para a ideia que você acabou de desenvolver.

Ele começa a dizer:

+Z ↑ │ ┌───────────┐ │ │ │ A0 │ │ 0,0,0 │ │ │ └───────────┘ ↙ ↘ -Y +X 

E aí você pode ter:

A-2 A-1 A0 A1 A2 │ B0 │ C0 

e, principalmente:

A0 = ( 0, 0, 0) A1 = ( 1, 0, 0) A2 = ( 2, 0, 0) B0 = ( 0, 1, 0) C0 = ( 0, 2, 0) A0+Z = (0, 0, 1) A0+Z2 = (0, 0, 2) 

Isso já é muito mais próximo do Index Space que você está descrevendo.

Mas tem uma coisa que eu mudaria

No segundo código você colocou:

Camada A = Núcleo
Camada B = raio 1
Camada C = raio 2

Eu não prenderia o Index Space a essa ideia de raio/concentricidade.

Porque seu modelo original é mais poderoso que isso.

Você não quer necessariamente:

C ┌─────┐ │ B │ │ ┌─┐ │ │ │A│ │ │ └─┘ │ └─────┘ 

Você quer poder fazer:

Z+ │ │ ┌──────┼──────┐ │ │ │ │ │ │ ───────┼──────A0─────┼────── X │ │ │ │ │ │ └──────┼──────┘ │ Y 

Ou seja:

o espaço é infinito/expansível em X, Y e Z.

A, B, C podem continuar sendo uma convenção visual sua, mas não devem limitar a matemática.

E aí os dois códigos se encaixam

Eu vejo a arquitetura assim:

INDEX SPACE (X, Y, Z) │ ▼ ┌───────────┐ │ CHUNK │ │ [x,y,z] │ └─────┬─────┘ │ ┌────────┼────────┐ ▼ ▼ ▼ cidade galpão fazenda │ ▼ conteúdo 

Então um chunk deixa de ser simplesmente:

chunk_03_07 

e passa a poder ser:

{ id: "chunk_03_07_02", x: 3, y: 7, z: 2 } 

E aí você pode colocar qualquer coisa dentro dele:

chunk (3,7,2) ├── chão ├── parede ├── porta ├── NPC ├── objeto └── conexão → chunk (4,7,2) 

E uma escada poderia fazer:

chunk (3,7,0) ↓ chunk (3,7,1) 

Uma ponte diagonal poderia fazer:

(3,7,0) ↘ (4,8,1) ↘ (5,9,1) 

Aí sim você está descrevendo uma trajetória dentro de um espaço volumétrico.

E o mais legal: seu primeiro código não é descartado. Ele vira uma das ferramentas que alimentam o Index Space.

O que eu faria agora é parar de mexer no mapa de Barueri por um instante e construir um protótipo mínimo do Index Space puro, sem cidade, sem estrada e sem geografia:

(0,0,0) no centro → chunks X/Y → empilhamento Z → conexões entre chunks.

Depois a gente joga o mapa 2D inteiro dentro dessa estrutura.

Sim. É exatamente esse o próximo passo. O protótipo atual está mostrando o espaço; agora ele precisa virar uma ferramenta de edição do espaço.

E eu faria uma mudança importante: esse painel que está permanentemente na frente não deveria ser o centro da interface. Ele pode virar uma ferramenta dentro de um Menu do Index Space.

Eu imaginaria assim

A tela fica praticamente limpa:

ZYRO — INDEX SPACE [0,0,1] ↑ │ ┌────┬────┬────┐ │ │ │ │ ├────┼────┼────┤ │ │ A0 │ │ ├────┼────┼────┤ │ │ │ │ └────┴────┴────┘ [ ☰ ] 

Você toca no ☰ e abre um menu radial.

Algo mais ou menos:

[ ↑ Z ] │ │ [ +Y ] ─ ● ─ [ +X ] │ [ −Z ] [CRIAR] [MOVER] [EXCLUIR] 

Mas eu não faria o radial somente para movimentação. Ele seria o menu de operações do Index Space.

1. Selecionar um chunk

Você toca em uma célula:

┌────┬────┬────┐ │ │ │ │ ├────┼────┼────┤ │ │ 🟧 │ │ ← selecionado ├────┼────┼────┤ │ │ │ │ └────┴────┴────┘ X: 0 Y: 0 Z: 0 

E o sistema sabe:

selected = { x: 0, y: 0, z: 0 } 

A partir daí, todas as ferramentas trabalham em relação à célula selecionada.

2. Criar espaço

Essa é uma das partes mais importantes da sua ideia.

Selecionou:

A0 = (0,0,0) 

Você abre o radial e escolhe:

CRIAR

Aí aparecem as direções:

↑ +Z │ +Y ← A0 → +X │ -Z ↓ 

Se tocar em +X:

A0 (0,0,0) → A1 (1,0,0) 

Se tocar em +Y:

A0 (0,0,0) → (0,1,0) 

Se tocar em +Z:

A0 (0,0,0) ↓ (0,0,1) 

E também:

-X -Y -Z 

Ou seja: você não está desenhando um cubo manualmente. Está expandindo o espaço através das coordenadas.

3. E aqui tem uma coisa ainda melhor

Eu colocaria duas operações diferentes:

ADICIONAR

Cria uma célula vizinha.

[0,0,0] +X ↓ [0,0,0] [1,0,0] 

EXPANDIR

Cria uma região inteira.

Por exemplo:

EXPANDIR X: -5 → +5 Y: -5 → +5 Z: 0 → 3 

Resultado:

┌───────────────┐ / /| / / | ┌───────────────┐ | │ │ | │ INDEX │ | │ SPACE │ / │ │ / └───────────────┘ 

Isso permite que você tenha um espaço enorme sem precisar criar manualmente cada chunk.

4. Depois entram as ferramentas de movimentação

Por exemplo:

MOVER

Você seleciona:

[3,7,0] 

e escolhe:

↑ +Z │ -X ← [●] → +X │ ↓ -Z 

Mover +X:

[3,7,0] ↓ [4,7,0] 

Mover +Z:

[3,7,0] ↓ [3,7,1] 

Mas eu colocaria uma regra fundamental:

Mover não deve destruir a identidade espacial.

O Index Space deveria distinguir entre:

posição

e

conteúdo.

Então você pode mover uma entidade:

ENTITY_001 de: [3,7,0] para: [4,7,0] 

sem alterar o sistema de coordenadas do espaço.

5. E eu adicionaria uma ferramenta chamada CONECTAR

Essa vai ser importante para aquilo que você estava falando da continuidade.

Você seleciona:

[3,7,0] 

depois:

CONECTAR 

e seleciona outro chunk:

[5,9,1] 

O Index Space registra:

[3,7,0] \ \ [4,8,1] \ \ [5,9,1] 

A conexão pode representar:

rua

ponte

escada

elevador

rampa

corredor

túnel

escada rolante

passagem

portal lógico

E isso é muito importante porque a conexão pertence ao espaço, não à câmera.

6. Eu mudaria o conceito do seu painel atual

O painel atual:

CONFIGURAÇÃO DO ESPAÇO Extensão Grid Empilhamento Z Chunks totais Eixo X Eixo Y Eixo Z 

não precisa desaparecer.

Ele vira uma ferramenta:

☰ │ ├── ESPAÇO │ ├── Criar │ ├── Expandir │ ├── Reduzir │ ├── CÉLULA │ ├── Selecionar │ ├── Mover │ ├── Duplicar │ ├── Excluir │ ├── CONEXÃO │ ├── Conectar │ ├── Desconectar │ ├── CAMADAS │ ├── Z+ │ ├── Z- │ └── Mostrar todas │ ├── VISUALIZAÇÃO │ ├── Grade │ ├── Eixos │ ├── Coordenadas │ └── Isométrico │ └── CONFIGURAÇÃO ├── Tamanho ├── Extensão └── Estatísticas 

Aí sim o Index Space começa a deixar de ser apenas um visualizador 3D e passa a ser um editor espacial.

E tem uma decisão arquitetural que eu acho fundamental

Eu separaria:

INDEX SPACE

da

CÂMERA.

A câmera pode estar:

ISOMÉTRICA TOP LATERAL FRONTAL FIRST PERSON ORBIT 

mas todas elas enxergam o mesmo espaço.

Então:

INDEX SPACE (X,Y,Z) │ ┌───────────┼───────────┐ ↓ ↓ ↓ TOP VIEW ISOMÉTRICO FIRST PERSON 

Isso casa perfeitamente com aquela ideia que você teve:

A câmera não define o mapa. O mapa define a câmera.

O protótipo que você enviou é **brilhante**. Ele resolve a complexidade da edição de um espaço 3D usando a metáfora de uma "Planta Baixa 2D Limpa".
**Por que isso funciona tão bem:**
 1. **Edição Simplificada:** Editar volume diretamente em 3D (especialmente com toques em telas móveis) é notoriamente difícil e impreciso. Sua abordagem 2D permite precisão cirúrgica na colocação de entidades e na definição de caminhos.
 2. **Abstração de "Planta Baixa":** Ao renderizar apenas retângulos coloridos e wireframes translúcidos, você foca na *organização espacial* (o "território"), e não na renderização visual final. Isso é o "The Sims/Modo Construtor" que você mencionou.
 3. **Persistência JSON:** A capacidade de exportar e importar o estado do território em JSON é fundamental. Isso prova que o mapa agora é uma estrutura de dados manipulável, e não apenas uma imagem renderizada.
### Implementando o Editor de Território Completo:
Conforme o seu roteiro, o próximo passo é evoluir o Sandbox 2D para o **Editor Index Space**, incorporando as ferramentas de Menu Radial, Seleção, Criação e Conexão.
#### 1. A Interface Limpa e o Menu Radial (☰)
O HUD permanentemente na frente sai de cena. A tela fica focada na planta baixa do grid. No canto, um único botão [ ☰ ].
Ao tocar nele, abre-se o **Menu Radial de Operações do Index Space**:
```
      [ ↑ Z ]      --- Ferramenta de Camada (Z+/Z-)
         │
[EXPANDIR]── ● ──[CONECTAR] --- Ferramentas de Espaço e Continuidade
         │
      [ ↓ Z ]
      
[CRIAR] [MOVER] [EXCLUIR] --- Ferramentas de Célula

```
Este menu não é apenas para movimentação, ele centraliza todas as ações editoriais do espaço.
#### 2. Seleção de Célula e Criação Expandida
**O fluxo de trabalho:**
 1. **Toque em uma Célula:** O sistema registra selected = { x: 0, y: 0, z: 0 } e destaca a célula.
 2. **Menu Radial \rightarrow CRIAR:** O menu radial muda para mostrar direções em relação à célula selecionada.
 3. **Toque em +X:** O sistema expande o espaço, criando a célula vizinha [1,0,0].
Este é o conceito fundamental: você **não desenha um cubo**, você **expande o espaço através das coordenadas**.
**A Ferramenta EXPANDIR:**
Para evitar o trabalho manual de criar chunk por chunk em áreas grandes, a ferramenta EXPANDIR permite definir uma região inteira:
EXPANDIR X: -5 → +5 | Y: -5 → +5 | Z: 0 → 3
Isso cria instantaneamente um volume imenso de chunks no Index Space, pronto para receber conteúdo.
#### 3. Conexões e a Continuidade Volumétrica
A ferramenta **CONECTAR** é o coração da continuidade espacial que você propôs.
**O fluxo:**
 1. **Selecionar Chunk A:** Toque em [3,7,0].
 2. **Menu Radial \rightarrow CONECTAR.**
 3. **Selecionar Chunk B:** Toque em [5,9,1].
O Index Space registra a conexão como uma entidade do espaço (não da câmera). A conexão pode representar qualquer estrutura de passagem: rua, escada, elevador, rampa, corredor, portal lógico, etc.
Isso é crucial: a conexão pertence ao território, permitindo que a "Ponte Diagonal" entre (3,7,0) e (5,9,1) exista matematicamente no Index Space, independentemente de como a câmera a visualiza.
#### 4. O Novo Papel do Painel de Configuração
O painel original não desaparece, ele vira uma ferramenta de diagnóstico dentro do menu:
☰ │ ├── CONFIGURAÇÃO ├── Tamanho ├── Extensão └── Estatísticas (Chunks Totais, Eixos)
### Conclusão e Próximo Protótipo
A arquitetura está sólida. Separar o **Index Space (Matemática)** do **Conteúdo (Dados)** e da **Câmera (Visualização)** é a decisão arquitetural correta.
A câmera é apenas um visualizador:
INDEX SPACE (X,Y,Z) │ ┌───────────┼───────────┐ ↓ ↓ ↓ Sandbox 2D Visualizador Isométrico Visualizador First Person
O mapa define a câmera; a câmera não define o mapa.
O próximo protótipo é exatamente este: o **Editor Index Space Completo**, integrando o Sandbox 2D com as ferramentas de Menu Radial, Seleção, Criação/Expansão e Conexão de células. Barueri, ruas e geografia só entram depois que essa fundação matemática estiver robusta.
É isso. E tem uma diferença importante agora: a gente já não está mais fazendo um “mapa 3D”. Estamos começando a fazer o editor do próprio espaço do ZYRO.

Eu seguiria exatamente nessa direção e faria o próximo protótipo como uma fusão dos dois códigos que você trouxe.

Estrutura do próximo protótipo

ZYRO — INDEX SPACE EDITOR │ ├── ESPAÇO │ ├── células [X,Y,Z] │ ├── expansão │ └── camadas │ ├── CONTEÚDO │ ├── entidades │ ├── construções │ └── objetos │ ├── RELAÇÕES │ └── conexões entre células │ └── VISUALIZAÇÃO ├── planta 2D ├── isométrico └── 3D 

E tem uma decisão que eu colocaria desde já no código:

A célula e a entidade são coisas diferentes

Uma célula pode existir vazia:

space["0,0,0"] = { x: 0, y: 0, z: 0 }; 

E depois receber conteúdo:

entities.push({ id: "entity_001", type: "residencia", position: { x: 0, y: 0, z: 0 } }); 

Isso significa que podemos ter:

INDEX SPACE [0,0,1] ← célula │ │ escada │ [0,0,0] ← célula │ 🏠 casa 

A casa não é o espaço.

A escada não é o espaço.

A célula não é a casa.

São três camadas diferentes.

E eu faria o radial com dois estados

Estado 1 — Menu principal

↑ CAMADA │ ESPAÇO ── ● ── CONECTAR │ CONTEÚDO [SELECIONAR] [CONFIG] 

Estado 2 — operação sobre célula

Depois de selecionar (0,0,0):

+Z ↑ [●] -X ← (0,0,0) → +X ↓ -Z ↑ +Y ↓ -Y 

Aí CRIAR significa literalmente:

criar o próximo espaço nessa direção.

E tem uma coisa que eu acrescentaria ao seu conceito de “EXPANDIR”

Não criaríamos necessariamente milhões de objetos JavaScript.

O Index Space pode ser esparso.

Se você disser:

X: -1000 → +1000 Y: -1000 → +1000 Z: 0 → 20 

não precisamos materializar 42 milhões de células imediatamente.

Podemos registrar:

space.bounds = { minX: -1000, maxX: 1000, minY: -1000, maxY: 1000, minZ: 0, maxZ: 20 }; 

E materializar chunks/células somente quando necessário.

Isso deixa o conceito de espaço expansível muito mais poderoso.

E o seu Sandbox 2D já fornece praticamente todo o mecanismo de interação que precisamos: pan, zoom, toque, seleção, ferramentas, entidades e JSON.

Então eu não faria um terceiro editor do zero.

Eu faria:

Sandbox 2D → Editor Index Space

adicionando:

coordenada Z;

seleção de célula;

criação de célula;

expansão;

exclusão;

conexões;

camadas;

menu radial;

JSON separado para space, entities e connections.

Existe uma parede naquele local?

A Gramática não inventa a geometria. Ela ajuda a determinar o que o usuário quis dizer.

A Gramática entende o mundo. O Index Space estrutura o mundo.

Essa separação é boa porque ela move o problema pra fora do lugar errado: hoje vocês estavam prestes a decidir estrutura espacial (`levels[]`, recursão) pra resolver um problema que é, na verdade, de **interpretação de linguagem**. "Segundo andar" nunca foi um conceito espacial — é um jeito humano de apontar pra um Z, e isso pode viver inteiramente na Gramática, sem contaminar o contrato do Index Space.

A trava que você colocou é a parte mais importante do desenho todo, então vale reforçar por que ela é necessária e não só desejável: se a Gramática pudesse escrever `Z = 1.7` diretamente, o Index Space deixaria de ser a fonte única de verdade — teria duas rotas pra chegar numa coordenada (uma validada pelo resolvedor, outra "direto da intenção"), e aí voltam os mesmos problemas de inconsistência que o v1 evitou ao fechar `pos+local` como única transformação. A trava garante que **só existe um caminho até uma coordenada válida**, não importa se a instrução veio de texto, de um clique no editor ou de um replay de log.

Isso também deixa mais claro o papel do "RESOLUÇÃO ESPACIAL" no meio do seu diagrama — pelo que você descreveu, ele não é uma terceira camada nova, é uma **API de consulta que o Index Space expõe** (existe `[4,7,1]`? há parede ali? o local está livre?), e a Gramática é cliente dessa API, nunca escritora direta de coordenada. Vale deixar isso explícito na mini-spec, porque "resolvedor espacial" soando como camada própria pode, sem querer, virar uma segunda fonte de verdade por acidente (alguém implementa o resolvedor com regras próprias de geometria em vez de delegar tudo pro Index Space).

1. Princípio

A Fase 2 é dividida em dois problemas independentes:

A) SEMÂNTICA DE LOCALIZAÇÃO "onde o humano quer dizer?" Gramática Zero ↓ consulta à API do Index Space ↓ coordenada válida B) ESTRUTURA ESPACIAL "o que fisicamente existe nesse espaço?" Index Space ↓ entidades / extensão / orientação / volumes 

A Gramática não altera o contrato espacial.

A Gramática interpreta.
O Index Space valida e representa.

2. Frente A — Semântica de localização

Status: ✅ RESOLVIDO CONCEITUALMENTE

Expressões humanas podem representar referências espaciais sem serem coordenadas.

Exemplos:

"segundo andar" "andar de cima" "na parede norte" "dentro da sala" "do outro lado da porta" "suba pela escada" "pegue o elevador para o terceiro andar" 

Essas expressões pertencem à Gramática Zero.

A Gramática produz uma intenção/referência sem escrever diretamente no espaço:

{ "intent": "place", "target": "porta", "location": { "semantic": "parede norte do segundo andar" } } 

Então consulta o Index Space:

Gramática Zero │ │ "segundo andar" ▼ API do Index Space │ ├── existe Z correspondente? ├── célula existe? ├── posição é válida? ├── existe parede? └── local está disponível? │ ▼ posição válida [pos + local] 

Regra inviolável

A Gramática não pode fazer isto:

Z = 1.7 

nem:

pos = [4,7,1.7] 

Ela só pode solicitar uma interpretação.

O Index Space é quem determina a representação espacial válida.

Os três testes passam, e o caso da ponte é o mais valioso dos três — não porque confirma a hipótese, mas porque ele expõe uma pergunta que os outros dois não expunham.

**Um jeito de tentar quebrar ainda mais o caso C:** se as células da ponte são contíguas e adjacentes umas às outras, então atravessar a ponte já é só "andar de célula em célula" — o mesmo mecanismo de movimento normal que funciona em qualquer chão. Isso levanta a pergunta: a ponte *precisa* de uma `connection` A→B, ou ela só precisa **existir como entidade ocupando células contíguas**, e a travessia acontece de graça porque adjacência já basta pra andar?

Se a resposta for "não precisa de connection nenhuma", isso na verdade fortalece ainda mais a arquitetura, porque revela pra que `connections` servem de verdade: **elas existem só quando A e B não são espacialmente adjacentes** (escada pulando Z, elevador saltando andares, atalho entre pontos distantes do mapa). Adjacência comum não precisa de connection — o motor de movimento já resolve isso olhando pras células vizinhas. Isso significa que o caso 9 (elevador) e o caso 2 (escada atravessando células) são os que realmente testam `connections`, enquanto a ponte na verdade testa só `cells[]` puro, sem tocar no outro conceito. Vale validar isso como um quarto teste formal: **"ponte contígua não precisa de connection; se precisar, a hipótese de adjacência automática está errada."**

**Sobre o `local` único vs. por célula** — acho que dá pra desenhar um teste específico em vez de decidir de cabeça: pega o prédio em L do caso A. Se ele for um bloco rígido (mesma altura, sem rotação variando célula a célula), um `local` âncora único já basta — cada célula ocupada é só "chão ocupado", sem posição interna própria. Agora pega uma ponte com **inclinação** (sobe de um lado pro outro, tipo rampa) — aí cada célula da ponte pode ter uma altura `z` de piso diferente, o que é literalmente um `local.z` diferente por célula. Se esse caso for real pro ZYRO (rampas, terrenos inclinados), a resposta vira "cada célula pode ter seu próprio local, com um default vazio/zero quando a entidade é rígida" — não é all-or-nothing, é: `local` por célula é opcional, mas quando ausente assume-se uma pose neutra derivada da própria célula.

Isso sugere que o próprio caso da ponte, se ela puder inclinar, já é o teste que decide essa pergunta — não precisa inventar um quarto caso separado.

**Sobre não adicionar `face` ainda** — concordo, e pelo mesmo princípio que vocês vêm aplicando: `rotation` já é suficiente pra geometria, e "parede norte" é reinterpretável pela Gramática a qualquer momento sem exigir mudança no schema — exatamente o padrão que a Frente A já resolveu pro "segundo andar".

Resumindo o que eu testaria antes de formalizar:
- [✓] prédio regular, [✓] prédio em L, [✓] torre descontínua — já couberam.
- [ ] ponte contígua *precisa* de connection, ou anda-se por adjacência sozinha?
- [ ] ponte com inclinação — decide se `local` é âncora única ou por-célula.

Se esses dois últimos também couberem sem regra nova, `cells[]` vira mesmo uma propriedade topológica só, e a Frente B fecha com dois conceitos (ocupação multi-célula + orientação de sub-célula), nenhum recursivo.

PRÉDIO ↓ cells[] PRÉDIO EM L ↓ cells[] TORRE ↓ cells[] ELEVADOR ↓ cells[] PONTE ↓ cells[] RAMPA ↓ cells[] + ? 

E, sobretudo:

X → Y X → Y → Z L vertical descontínuo 

não exigem estruturas diferentes.

Isso é exatamente o que queríamos encontrar.

E eu faria uma pequena mudança na definição da Frente B

Em vez de dizer:

"Entidade referencia mais de uma célula."

eu escreveria:

"Uma entidade pode possuir uma ocupação espacial composta por uma ou mais células do Index Space."

Porque isso abre espaço para descobrirmos que cells[] talvez não seja simplesmente um array de coordenadas.

E mantém o conceito independente da geometria.

Estado atual

FASE 2 · FRENTE B MULTI-CÉLULA [✓] bloco regular [✓] forma em L [✓] ocupação vertical [✓] ocupação descontínua [✓] ponte contígua CONNECTION [✓] não necessária para adjacência [ ] necessária para salto não-adjacente [ ] elevador [ ] portal [ ] escada não-adjacente GEOMETRIA LOCAL [✓] entidade rígida → local único parece suficiente [ ] rampa inclinada [ ] superfície com variação por célula ORIENTAÇÃO [✓] rotation existente [✓] "norte/sul" pode continuar semântico [ ] nenhum novo campo necessário por enquanto RECURSÃO [✓] nenhum caso exigiu 

E tem uma consequência arquitetural muito bonita aqui:

O mapa não precisa registrar todas as possibilidades de movimento como connections.

Ele registra estrutura e exceções.

O motor deriva o que é óbvio:

célula A ↓ vizinhança ↓ célula B ↓ movimento normal 

E consulta connections somente quando existe algo que não pode ser inferido da estrutura local.

Isso deixa o Index Space ainda mais parecido com aquilo que estamos tentando construir desde o começo: um espaço matemático mínimo, do qual o comportamento pode ser derivado, em vez de um mapa cheio de instruções explícitas.

A distinção entre topologia implícita (adjacência) e relação explícita (`connection`) é exatamente o tipo de princípio que vale "congelar" — ela resolve de uma vez o que antes parecia caso a caso (ponte vs elevador vs escada) com uma única regra de decisão: *dá pra chegar lá andando célula a célula? Então não precisa de connection.*

Sobre o impasse do `cells[]` ter dois formatos — acho que acharam o problema certo, mas talvez a saída não seja escolher entre "lista simples" ou "lista de objetos", e sim **separar onde a variação mora**, do mesmo jeito que vocês já separaram Gramática de Index Space. Uma opção que evita os dois formatos concorrentes:

- `cells` continua **sempre** uma lista pura de coordenadas — puramente topológica, nunca muda de forma, seja pra prédio, torre, ponte ou rampa. Ela responde só "o que essa entidade ocupa".
- Um campo **separado e opcional**, algo como `local_overrides` (mapeado por célula), aparece só quando a entidade precisa de geometria variável célula a célula. Quando ausente pra uma célula, usa-se o `local` único da entidade como default.

Isso preserva a ideia que vocês mesmos levantaram ("separar ocupação de geometria"), mas sem empurrar a variação pro `type` — a rampa não precisa de uma regra especial de "type: rampa infere progressão"; ela só declara explicitamente `local_overrides` pras três células, e a entidade rígida simplesmente nunca usa esse campo. Isso mantém uma única forma canônica pra `cells` (nunca dois formatos concorrentes pro mesmo array) e isola a exceção (geometria variável) num campo que só existe quando é preciso — o que é consistente com o espírito de "registrar estrutura e exceções" que vocês descreveram pro `connections`.

Um teste de falsificação que eu tentaria antes de fechar isso: uma rampa **em espiral**, que não só varia `z` mas também gira (`rotation` mudando célula a célula, não só altura). Se `local_overrides` continuar dando conta disso sem precisar tocar em `rotation` no nível da entidade, a hipótese fica ainda mais forte — porque prova que o campo aguenta variação de orientação, não só de altura, sem exigir um terceiro mecanismo.

Quanto ao estado geral: acho que a Frente B já tem evidência suficiente pra fechar **ocupação multi-célula** (`cells[]` puro) como resolvido, e deixar só **geometria variável por célula** como a última peça em teste — o que é um escopo bem menor do que "volume e orientação" parecia no início.

Registrado. Se quiser, na próxima rodada dá pra atacar o teste da rampa em espiral pra fechar a última peça em aberto da Frente B.

## ⚠️ Problemas Detectados
 * **Lógica de Negócio e Renderização Misturadas**: No arquivo do Editor Canvas 2D e no protótipo Three.js, os scripts contêm tanto a definição dos dados (TYPES, GEO_NODES) quanto a lógica de manipulação da UI, eventos de mouse e o loop de renderização (requestAnimationFrame).
 * **Uso de Globais no Escopo de Window**: O estado global (entities, activeTool, selectedEntity, camera) fica exposto diretamente na janela, o que dificulta testes automatizados e favorece colisão de estados.
 * **Falta de Abstração para o Engine do Index Space**: Embora a documentação conceitual especifique a separação entre a API do Index Space e os Renderers, os protótipos de código injetam posições diretamente dentro das estruturas visuais sem passar por um resolvedor/API unificada.
 * **Inconsistência entre os Modelos de Coordenadas**: O protótipo 2D Canvas trabalha com gx, gy (2D), enquanto o arquivo Three.js projeta XZ no plano 3D, e a especificação define XYZ + local. Não há uma biblioteca central de conversão reutilizada pelos dois frontends.

 2. **Implementação do Padrão Adapter/Renderer**: Criar contratos de renderização claros para que tanto a visão 2D (Canvas) quanto a 3D (Three.js) sejam apenas leitores reativos do mesmo estado mantido pelo Index Space Core.
 3. **Separação em Módulos (ES Modules)**:
   * src/core/indexspace.js: Estrutura de coordenadas, validação e manipulação do estado.
   * src/core/grammar.js: Resolvedor semântico (Gramática Zero).
   * src/renderers/canvas2d.js: Adapter para a planta baixa.
   * src/renderers/three3d.js: Adapter para o grid 3D.
   * src/ui/hud.js: Manipulação dos eventos do DOM e ferramentas do usuário.
 4. **Validação Rigorosa de Schemas**: Implementar a validação das estruturas JSON de entrada e saída contra o schema oficial zyro.indexspace.v1 antes de aplicar as atualizações no canvas.

### Outros Blocos (Especificação e Invariantes da Spec v2)
```markdown
3. Entidade multi-célula
Regra: cells é sempre array de coordenadas [x,y,z]. Nunca objetos.
Sua função é exclusivamente: declarar a ocupação topológica da entidade. Não representa geometria interna.

4. Geometria padrão
A entidade possui uma geometria padrão: local, rotation.
Quando todas as células compartilham essa geometria, nada mais é necessário.

5. Variação geométrica por célula
Quando uma ou mais células precisam fugir do padrão: cell_overrides.

∀ k ∈ keys(cell_overrides): parse(k) ∈ cells

7. Adjacência
A adjacência entre células é implícita e não precisa ser registrada em connections.

8. Connections
connections fica reservado para relações espaciais não deriváveis da adjacência normal (elevador, portal, teletransporte).

CONNECTION → explícita
```

### Fluxo de Funcionamento e Conexão de Conceitos
 * **Isolamento de Camadas Espaciais**: O Index Space atua como a única fonte da verdade topológica e geométrica. A Gramática Zero traduz linguagem natural/semântica ("segundo andar") em intenções de consulta sem permissão para mutação direta das coordenadas. O Renderer lê o estado do Index Space para desenhar a cena (2D/3D).
 * **Representação Topológica de Entidades**:
   * Entidades de célula única usam pos, local e rotation.
   * Entidades multi-célula utilizam o array cells (contendo tuplas de coordenadas [x,y,z]).
   * Exceções locais são tratadas via dicionário cell_overrides.
 * **Validação e Integridade (Fail Closed)**: A regra de validação \forall k \in \text{keys}(cell\_overrides): \text{parse}(k) \in cells garante que alterações de geometria por célula só existam para células explicitamente mapeadas em cells. Se um override referenciar uma célula inexistente na entidade, a operação falha imediatamente.
 * **Topologia de Grafos e Conectividade**: Adjacência cartesiana imediata é calculada por derivação matemática sem estado persistido. O campo connections é mantido exclusivamente para arestas com descontinuidade espacial (teleportes, portais, elevadores).

// ============================================================
// ZYRO · INDEX SPACE v2 — VALIDADOR ESTRUTURAL
// ============================================================
//
// Responsabilidade única: integridade estrutural do contrato espacial.
// Não sabe o que é uma ponte, prédio, rampa ou torre.
// Não decide navegabilidade (isso é NavigationValidator, outra camada).
// Não decide geometria "correta" (isso é GeometryValidator, outra camada).
// Só verifica: célula, coordenada, ocupação, referência, integridade.
//
// Invariantes verificadas:
// 1. cells (quando presente) é array plano de [x,y,z] — nunca objetos
// 2. cells não tem coordenadas duplicadas
// 3. toda coordenada em cells existe em space.cells (existência estrutural)
// 4. pos existe em space.cells
// 5. cells presente => pos ∈ cells
// 6. cell_overrides presente => toda chave ∈ cells (parse(k) ∈ cells)
//
// Semântica de ausência:
// cells ausente → entidade ocupa somente pos
// cells presente → entidade ocupa exatamente cells
// ============================================================

function coordKey(coord) {
// Normaliza [x,y,z] -> "x,y,z", igual ao formato de chave usado em cell_overrides
return coord.join(',');
}

function parseKey(key) {
// "x,y,z" -> [x,y,z] (números), o inverso de coordKey
return key.split(',').map(Number);
}

function cellExists(coord, spaceCells) {
// space.cells no schema v1 é esparso, chaveado por "x,y,z"
return Object.prototype.hasOwnProperty.call(spaceCells, coordKey(coord));
}

/**

Valida a estrutura espacial de uma entidade contra o Index Space.

@param {object} entity - entidade com pos, cells? e cell_overrides? opcionais

@param {object} spaceCells - world.space.cells (mapa esparso "x,y,z" -> {})

@returns {{ valid: boolean, errors: string[] }}
*/
function validateEntity(entity, spaceCells) {
const errors = [];
const { pos, cells, cell_overrides } = entity;

// --- pos é obrigatório e deve existir no espaço ---
if (!pos || !Array.isArray(pos) || pos.length !== 3) {
errors.push(pos ausente ou malformado: ${JSON.stringify(pos)});
} else if (!cellExists(pos, spaceCells)) {
errors.push(pos ${JSON.stringify(pos)} não existe em space.cells);
}

// --- cells, quando presente, define a ocupação exata ---
if (cells !== undefined) {
if (!Array.isArray(cells)) {
errors.push(cells deve ser um array plano de coordenadas, recebeu: ${typeof cells});
} else {
const seen = new Set();
let hasMalformed = false;

for (const c of cells) { // formato: cada item deve ser [x,y,z], nunca {pos, local} if (!Array.isArray(c) || c.length !== 3 || c.some(n => typeof n !== 'number')) { errors.push(`item de cells não é uma coordenada [x,y,z] válida: ${JSON.stringify(c)}`); hasMalformed = true; continue; } const key = coordKey(c); // duplicata if (seen.has(key)) { errors.push(`cells contém coordenada duplicada: ${JSON.stringify(c)}`); } seen.add(key); // existência estrutural (não navegabilidade) if (!cellExists(c, spaceCells)) { errors.push(`célula ${JSON.stringify(c)} declarada em cells não existe em space.cells`); } } // pos ∈ cells if (!hasMalformed && pos && Array.isArray(pos) && pos.length === 3) { const posInCells = cells.some(c => coordKey(c) === coordKey(pos)); if (!posInCells) { errors.push(`pos ${JSON.stringify(pos)} não está contido em cells — âncora fora da ocupação`); } } } 

}

// --- cell_overrides, quando presente, só pode referenciar células ocupadas ---
if (cell_overrides !== undefined) {
if (typeof cell_overrides !== 'object' || cell_overrides === null || Array.isArray(cell_overrides)) {
errors.push(cell_overrides deve ser um objeto chave->override, recebeu: ${typeof cell_overrides});
} else {
// universo de células válidas: cells se presente, senão só pos
const occupied = new Set(
(cells !== undefined && Array.isArray(cells))
? cells.map(coordKey)
: (pos ? [coordKey(pos)] : [])
);

for (const key of Object.keys(cell_overrides)) { if (!occupied.has(key)) { errors.push(`cell_overrides tem chave "${key}" que não corresponde a nenhuma célula em cells (órfã)`); } } } 

}

return { valid: errors.length === 0, errors };
}

**Casos mínimos de navegação (candidatos)**

1. **Chão livre, sem entidades** — `[0,0,0] → [1,0,0]`, adjacentes, nada ocupando nenhuma das duas. Baseline: deve passar.
2. **Parede entre duas células adjacentes** — a parede (já modelada como entidade com `pos`+`rotation`) fica *entre* A e B. Aqui a pergunta interessante é: o bloqueio pertence à **aresta** A↔B, ou à célula que contém a parede? Se a parede é uma entidade dentro de uma célula só, isso sugere que bloqueio não é propriedade da célula inteira, é propriedade de uma face/direção dela.
3. **Porta na mesma posição da parede** — mesma aresta do caso 2, mas com porta permitindo passagem. Testa se bloqueio é *derivado da entidade presente*, não de um campo estático tipo `walkable`.
4. **Obstáculo ocupando a célula inteira** (ex: rocha, pilar sólido) — diferente do caso 2: aqui não é a aresta que bloqueia, é a célula-destino inteira que se torna intransitável, mesmo sendo adjacente e existente. Isso separa dois mecanismos de bloqueio (aresta vs. célula).
5. **Ponte contígua** (já resolvido) — entidade ocupando várias células adjacentes, sem bloquear nem exigir connection; prova que ocupação ≠ bloqueio por padrão.
6. **Rampa contígua com `cell_overrides`** — mesma lógica da ponte, mas variando altura; testa se `cell_overrides` (que é puramente geométrico) não interfere em nada na navegação — navegação devia ser cega a isso.
7. **Escada como connection não-adjacente** (já coberto pela distinção adjacência≠connection) — `[1,0,0] → [1,0,1]` sem serem vizinhas no grid, só navegável via `connections`.
8. **Elevador com salto grande de Z** — mesmo mecanismo do caso 7, prova que `connections` não tem restrição de "distância pequena".
9. **Passagem de mão única** (ex: catraca, borda de penhasco) — `A → B` permitido, `B → A` bloqueado. Testa se a transição precisa ser modelada como **direcional**, não simétrica — isso quebraria a suposição implícita de `isAdjacent()` de que adjacência (e navegabilidade) é sempre bidirecional.
10. **Célula ausente do `space.cells` esparso** (já é território do StructuralValidator — só incluo pra confirmar que Navigation nem chega a rodar aí, `validateNavigation` já retorna erro antes de perguntar sobre bloqueio).

function coordKey(coord) {
  return coord.join(',');
}

function cellExists(coord, spaceCells) {
  return Object.prototype.hasOwnProperty.call(spaceCells, coordKey(coord));
}

function isAdjacent(a, b) {
  const dx = Math.abs(a[0] - b[0]);
  const dy = Math.abs(a[1] - b[1]);
  const dz = Math.abs(a[2] - b[2]);
  return dx + dy + dz === 1;
}

// Procura uma connection EXATA e DIRECIONAL: from->to, nesta ordem.
// Não existe fallback simétrico — se quiser B->A, precisa de outra connection.
function findConnection(connections, from, to) {
  return connections.find(
    c => coordKey(c.from) === coordKey(from) && coordKey(c.to) === coordKey(to)
  );
}

/**
 * Harness v0 — deliberadamente ignora entidades (parede/porta/obstáculo/
 * ponte/rampa/elevador). Só enxerga: espaço estrutural + connections.
 */
function validateNavigation(world, from, to) {
  const spaceCells = world.space.cells;

  if (!cellExists(from, spaceCells)) {
    return { valid: false, reason: 'structural_error', errors: [`origem ${JSON.stringify(from)} não existe`] };
  }
  if (!cellExists(to, spaceCells)) {
    return { valid: false, reason: 'structural_error', errors: [`destino ${JSON.stringify(to)} não existe`] };
  }

  if (isAdjacent(from, to)) {
    return { valid: true, reason: 'adjacent' };
  }

  const connection = findConnection(world.connections || [], from, to);
  if (connection) {
    return { valid: true, reason: 'connection', connection };
  }

  return { valid: false, reason: 'no_path', errors: [`${JSON.stringify(from)} → ${JSON.stringify(to)} não é adjacente e não há connection`] };
}

// A implementação é deliberadamente burra:
//   1. verifica existência de origem/destino no espaço
//   2. verifica adjacência topológica (Manhattan distance == 1)
//   3. procura uma connection explícita e DIRECIONAL (from -> to)
//   4. NÃO sabe o que é parede, porta, obstáculo, ponte, rampa ou elevador
//   5. registra o mecanismo pelo qual passou (reason), não só valid/invalid
//
// Objetivo: deixar os 10 casos comprovarem ou quebrarem as hipóteses H1-H10,
// em vez de decidir o schema de navegação antes da evidência.
//
// H1  — adjacência cria movimento potencial
// H2  — adjacência não implica necessariamente passagem (esperamos que
//        esta hipótese REVELE UM GAP: o harness v0 ainda não modela isso)
// H3  — bloqueio pode pertencer à célula (obstáculo) — GAP esperado
// H4  — bloqueio pode pertencer à fronteira entre células (parede) — GAP esperado
// H5  — ocupação de célula não implica bloqueio (ponte/rampa) — deve PASSAR
// H6  — geometria (local/rotation/cell_overrides) é invisível pra navegação
// H7  — connection permite passagem não-adjacente
// H8  — connection é direcional por natureza: from -> to, sem precisar de oneWay
// H9  — não existe walkable declarado em cell até um caso provar necessidade
// H10 — NavigationValidator trabalha sobre espaço estruturalmente válido
// ============================================================

function coordKey(coord) {
  return coord.join(',');
}

function cellExists(coord, spaceCells) {
  return Object.prototype.hasOwnProperty.call(spaceCells, coordKey(coord));
}

function isAdjacent(a, b) {
  const dx = Math.abs(a[0] - b[0]);
  const dy = Math.abs(a[1] - b[1]);
  const dz = Math.abs(a[2] - b[2]);
  return dx + dy + dz === 1;
}

// Procura uma connection EXATA e DIRECIONAL: from->to, nesta ordem.
// Não existe fallback simétrico — se quiser B->A, precisa de outra connection.
function findConnection(connections, from, to) {
  return connections.find(
    c => coordKey(c.from) === coordKey(from) && coordKey(c.to) === coordKey(to)
  );
}

/**
 * Harness v0 — deliberadamente ignora entidades (parede/porta/obstáculo/
 * ponte/rampa/elevador). Só enxerga: espaço estrutural + connections.
 */
function validateNavigation(world, from, to) {
  const spaceCells = world.space.cells;

  if (!cellExists(from, spaceCells)) {
    return { valid: false, reason: 'structural_error', errors: [`origem ${JSON.stringify(from)} não existe`] };
  }
  if (!cellExists(to, spaceCells)) {
    return { valid: false, reason: 'structural_error', errors: [`destino ${JSON.stringify(to)} não existe`] };
  }

  if (isAdjacent(from, to)) {
    return { valid: true, reason: 'adjacent' };
  }

  const connection = findConnection(world.connections || [], from, to);
  if (connection) {
    return { valid: true, reason: 'connection', connection };
  }

  return { valid: false, reason: 'no_path', errors: [`${JSON.stringify(from)} → ${JSON.stringify(to)} não é adjacente e não há connection`] };
}

Como [20,0,0] e [21,0,0] são adjacentes (distância Manhattan = 1), o harness resolve a passagem pela regra de adjacência antes de sequer olhar pra connections — então a direção do from/to da connection nunca chega a ser consultada nos dois sentidos. O resultado: 9a passa (correto, mas por acidente) e 9b também passa (errado) — porque adjacência concede passagem incondicional, e a connection nunca entra em jogo quando as células já são vizinhas.

Isso muda a leitura do H8: a hipótese "from→to já basta pra codificar direção, sem precisar de oneWay" continua válida só pra passagens não-adjacentes (escada, elevador — onde a adjacência não interfere). Mas passagem restrita entre células adjacentes — como a catraca — não é um problema de "direção de connection", é o mesmo problema que a parede e o obstáculo: é bloqueio, só que direcional em vez de total. 9b deveria ter entrado no Bloco B (bloqueio), não no Bloco D (connection) — e o harness só revelou isso porque insistimos em testar com fixtures adjacentes de verdade, em vez de assumir que "direcionalidade" seria sempre resolvida por connection.
